.PHONY: help install dev test lint format run run-dry init-db clean docker-up docker-down

help:
	@echo "Commandes disponibles :"
	@echo "  make install      - Installer les dépendances"
	@echo "  make dev          - Setup développement"
	@echo "  make test         - Lancer les tests"
	@echo "  make lint         - Vérifier le code"
	@echo "  make format       - Formater le code"
	@echo "  make run          - Exécuter le pipeline"
	@echo "  make run-dry      - Exécuter en mode simulation"
	@echo "  make init-db      - Initialiser la base de données"
	@echo "  make clean        - Nettoyer les artefacts"
	@echo "  make docker-up    - Démarrer PostgreSQL"
	@echo "  make docker-down  - Arrêter PostgreSQL"

install:
	python -m venv .venv
	.venv/bin/pip install --upgrade pip
	.venv/bin/pip install -r requirements.txt

dev: install
	.venv/bin/pip install pytest pytest-cov black flake8 mypy

test:
	.venv/bin/pytest tests/ -v --cov=. --cov-report=html

lint:
	.venv/bin/flake8 . --count --select=E9,F63,F7,F82 --show-source --statistics
	.venv/bin/mypy . --ignore-missing-imports

format:
	.venv/bin/black . --line-length=100

run: init-db
	.venv/bin/python app.py --run-now --verbose

run-dry:
	.venv/bin/python app.py --run-now --dry-run --verbose

init-db:
	.venv/bin/python app.py --init-db

clean:
	find . -type d -name __pycache__ -exec rm -rf {} +
	find . -type f -name "*.pyc" -delete
	find . -type f -name "*.pyo" -delete
	rm -rf .pytest_cache/
	rm -rf htmlcov/
	rm -rf .coverage

docker-up:
	docker-compose up -d

docker-down:
	docker-compose down

docker-logs:
	docker-compose logs -f postgres
